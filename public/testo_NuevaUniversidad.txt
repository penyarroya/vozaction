package com.unilab.campus.audit;

import jakarta.persistence.*;
import lombok.Getter;
import lombok.Setter;
import org.springframework.data.annotation.CreatedDate;
import org.springframework.data.annotation.LastModifiedDate;
import org.springframework.data.jpa.domain.support.AuditingEntityListener;

import java.time.LocalDateTime;

@Getter
@Setter
@MappedSuperclass
@EntityListeners(AuditingEntityListener.class)
public abstract class Auditable {

    @CreatedDate
    @Column(name = "created_at", nullable = false, updatable = false)
    private LocalDateTime createdAt;

    @LastModifiedDate
    @Column(name = "updated_at")
    private LocalDateTime updatedAt;
}

package com.unilab.campus.entity;

import com.unilab.campus.audit.Auditable;
import jakarta.persistence.*;
import lombok.*;
import org.hibernate.annotations.UuidGenerator;

import java.util.ArrayList;
import java.util.List;

@Entity
@Table(name = "institutions")
@Getter
@Setter
@NoArgsConstructor
@AllArgsConstructor
@Builder
public class Institution extends Auditable {

    @Id
    @UuidGenerator
    @Column(name = "id", nullable = false, updatable = false)
    private String id;

    @Column(name = "name", nullable = false)
    private String name;

    @Column(name = "description", columnDefinition = "TEXT")
    private String description;

    @Column(name = "logo")
    private String logo;

    @Column(name = "website")
    private String website;

    @Column(name = "email", nullable = false)
    private String email;

    @Column(name = "phone")
    private String phone;

    @Column(name = "address")
    private String address;

    @Column(name = "allow_public_registration")
    @Builder.Default
    private Boolean allowPublicRegistration = true;

    @Column(name = "default_role")
    @Builder.Default
    private String defaultRole = "STUDENT";

    @Column(name = "theme")
    private String theme;

    @Column(name = "is_active")
    @Builder.Default
    private Boolean isActive = true;

    @OneToMany(mappedBy = "institution", cascade = CascadeType.ALL, fetch = FetchType.LAZY)
    private List<Course> courses = new ArrayList<>();

    @OneToMany(mappedBy = "institution", cascade = CascadeType.ALL, fetch = FetchType.LAZY)
    private List<UserEntity> users = new ArrayList<>();
}

package com.unilab.campus.entity;

import com.unilab.campus.audit.Auditable;
import com.unilab.campus.enums.CourseLevel;
import com.unilab.campus.enums.Visibility;
import jakarta.persistence.*;
import lombok.*;
import org.hibernate.annotations.UuidGenerator;

import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.List;

@Entity
@Table(name = "courses")
@Getter
@Setter
@NoArgsConstructor
@AllArgsConstructor
@Builder
public class Course extends Auditable {

    @Id
    @UuidGenerator
    @Column(name = "id", nullable = false, updatable = false)
    private String id;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "institution_id", nullable = false)
    private Institution institution;

    @Column(name = "title", nullable = false)
    private String title;

    @Column(name = "description", columnDefinition = "TEXT")
    private String description;

    @Enumerated(EnumType.STRING)
    @Column(name = "level")
    private CourseLevel level;

    @Column(name = "is_active")
    @Builder.Default
    private Boolean isActive = true;

    @ElementCollection
    @CollectionTable(name = "course_categories", joinColumns = @JoinColumn(name = "course_id"))
    @Column(name = "category")
    private List<String> categories = new ArrayList<>();

    @ElementCollection
    @CollectionTable(name = "course_tags", joinColumns = @JoinColumn(name = "course_id"))
    @Column(name = "tag")
    private List<String> tags = new ArrayList<>();

    @Column(name = "language")
    @Builder.Default
    private String language = "es-ES";

    @Column(name = "estimated_hours")
    private Integer estimatedHours;

    @Column(name = "max_weeks")
    private Integer maxWeeks;

    @Enumerated(EnumType.STRING)
    @Column(name = "visibility")
    @Builder.Default
    private Visibility visibility = Visibility.PUBLIC;

    @Column(name = "enrollment_type")
    @Builder.Default
    private String enrollmentType = "OPEN";

    @Column(name = "total_enrollments")
    @Builder.Default
    private Integer totalEnrollments = 0;

    @Column(name = "total_completions")
    @Builder.Default
    private Integer totalCompletions = 0;

    @Column(name = "average_progress")
    @Builder.Default
    private Double averageProgress = 0.0;

    @Column(name = "rating")
    @Builder.Default
    private Double rating = 0.0;

    @Column(name = "start_date")
    private LocalDateTime startDate;

    @Column(name = "end_date")
    private LocalDateTime endDate;

    @Column(name = "created_by")
    private String createdBy;

    @OneToMany(mappedBy = "course", cascade = CascadeType.ALL, fetch = FetchType.LAZY)
    private List<Collection> collections = new ArrayList<>();

    @OneToMany(mappedBy = "course", cascade = CascadeType.ALL, fetch = FetchType.LAZY)
    private List<Enrollment> enrollments = new ArrayList<>();
}

package com.unilab.campus.entity;

import com.unilab.campus.audit.Auditable;
import jakarta.persistence.*;
import lombok.*;
import org.hibernate.annotations.UuidGenerator;

import java.util.ArrayList;
import java.util.List;

@Entity
@Table(name = "collections")
@Getter
@Setter
@NoArgsConstructor
@AllArgsConstructor
@Builder
public class Collection extends Auditable {

    @Id
    @UuidGenerator
    @Column(name = "id", nullable = false, updatable = false)
    private String id;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "course_id", nullable = false)
    private Course course;

    @Column(name = "title", nullable = false)
    private String title;

    @Column(name = "description", columnDefinition = "TEXT")
    private String description;

    @Column(name = "icon")
    private String icon;

    @Column(name = "display_order")
    private Integer order;

    @Column(name = "is_required")
    @Builder.Default
    private Boolean isRequired = true;

    @Column(name = "estimated_hours")
    private Integer estimatedHours;

    @OneToMany(mappedBy = "collection", cascade = CascadeType.ALL, fetch = FetchType.LAZY)
    @OrderBy("order ASC")
    private List<Topic> topics = new ArrayList<>();
}

@Entity

@Getter
@Setter
@NoArgsConstructor

@Builder

    @Id
    @UuidGenerator
    @Column(name = "id", nullable = false, updatable = false)

📚 NIVEL 1: INSTITUCIÓN
¿Qué es?
Es la organización educativa que ofrece los cursos. Permite tener múltiples instituciones en la misma plataforma.

Ejemplo en código:
json
{
  "id": "inst-001",
  "name": "JLRN - Universilab Ciudadana",
  "description": "Centro de inclusión digital para personas sin experiencia tecnológica",
  "email": "info@universilab.org",
  "website": "https://universilab.org",
  "allowPublicRegistration": true
}
¿Para qué sirve?
Agrupa todos los cursos de una institución

Configuración global (tema, registro, roles)

Multi-tenant: varias instituciones en una misma plataforma

📚 NIVEL 2: CURSO
¿Qué es?
Es un programa educativo completo con una estructura de aprendizaje. Puede tener muchos estudiantes y mentores.

Ejemplo en código:
json
{
  "id": "course-001",
  "institutionId": "inst-001",
  "title": "★ INFORMÁTICA POPULAR ★ Inclusión desde Cero",
  "description": "Aprende informática desde cero, sin conocimientos previos. Diseñado para personas con poca experiencia digital.",
  "level": "BEGINNER",
  "isActive": true,
  "visibility": "PUBLIC",
  "language": "es-ES",
  "estimatedHours": 40,
  "categories": ["Tecnología", "Inclusión Digital"]
}
¿Para qué sirve?
Define el programa de aprendizaje

Establece el nivel (principiante, intermedio, avanzado)

Controla visibilidad (público, privado, no listado)

Métricas: número de inscritos, completados, valoración

👥 NIVEL 3: INSCRIPCIONES (ENROLLMENT)
¿Qué es?
Es el vínculo entre un usuario y un curso. Define qué rol tiene dentro del curso (estudiante, mentor, colaborador).

Roles disponibles:
Rol	Qué puede hacer
LEARNER	Ver contenido, hacer progreso, comentar
MENTOR	Guiar, responder dudas, evaluar
COLLABORATOR	Colaborar en la creación de contenido
Ejemplo en código:
json
{
  "id": "enroll-001",
  "courseId": "course-001",
  "userId": "user-123",
  "role": "LEARNER",
  "status": "ACTIVE",
  "overallProgress": 65.5,
  "timeSpent": 240,
  "enrolledAt": "2026-08-15T10:00:00",
  "lastAccess": "2026-08-23T18:30:00"
}
Estados de inscripción:
PENDING: Pendiente de aprobación

ACTIVE: Activo, puede acceder

COMPLETED: Curso completado

DROPPED: Abandonó

SUSPENDED: Suspendido temporalmente

📂 NIVEL 4: COLECCIÓN
¿Qué es?
Es un agrupador de temas dentro de un curso. Ayuda a organizar el contenido por módulos o áreas temáticas.

Ejemplo en código:
json
{
  "id": "coll-001",
  "courseId": "course-001",
  "title": "★ Tecnología sin Barreras ★",
  "description": "Fundamentos básicos de tecnología con un enfoque inclusivo",
  "order": 1,
  "isRequired": true,
  "estimatedHours": 10
}
¿Para qué sirve?
Organiza el contenido en módulos

Permite marcar módulos como obligatorios u opcionales

Define el orden de aprendizaje

📖 NIVEL 5: TEMA (TOPIC)
¿Qué es?
Es una unidad temática dentro de una colección. Agrupa subtemas relacionados.

Ejemplo en código:
json
{
  "id": "topic-001",
  "collectionId": "coll-001",
  "title": "Introducción a la Informática",
  "description": "Conceptos básicos y fundamentos de la informática",
  "order": 1,
  "estimatedMinutes": 45,
  "isPublished": true
}
¿Para qué sirve?
Divide el contenido en unidades temáticas

Establece el orden de aprendizaje

Permite publicar/despublicar contenido

📑 NIVEL 6: SUBTEMA (SUBTOPIC)
¿Qué es?
Es la unidad de contenido más detallada. Aquí se encuentra el contenido educativo real y los recursos asociados.

Estructura completa:
json
{
  "id": "subtopic-001",
  "topicId": "topic-001",
  "title": "Historia de la Informática",
  "content": "<h1>Historia de la Informática</h1><p>La informática tiene sus orígenes...</p>",
  "summary": "Breve recorrido por la historia de la computación",
  "order": 1,
  "estimatedMinutes": 15,
  "authorId": "user-456",
  "authorName": "María González",
  "accessibility": {
    "hasAudioDescription": true,
    "hasSignLanguage": false,
    "hasTranscript": true,
    "hasAlternativeText": true
  }
}
¿Qué contiene un subtema?
Contenido educativo (texto, imágenes, videos)

Recursos (PDFs, videos, enlaces)

Comentarios (discusión entre usuarios)

Contribuciones (sugerencias de mejora)

Progreso (seguimiento de cada usuario)

📄 NIVEL 7: PÁGINA (PAGE)
¿Qué es?
Es la unidad más pequeña de contenido. Una página es un elemento individual dentro de un subtema.

Tipos de páginas:
Tipo	Descripción
TEXT	Contenido textual con imágenes
VIDEO	Video educativo
QUIZ	Cuestionario interactivo
EXERCISE	Ejercicio práctico
INTERACTIVE	Contenido interactivo
Ejemplo en código:
json
{
  "id": "page-001",
  "subtopicId": "subtopic-001",
  "title": "Orígenes y fundamentos",
  "content": "<p>La informática como disciplina...</p>",
  "type": "TEXT",
  "order": 1,
  "estimatedMinutes": 10,
  "isPublished": true
}
📎 RECURSOS (RESOURCE)
¿Qué es?
Son archivos multimedia asociados a subtemas o páginas.

Tipos de recursos:
Tipo	Ejemplos
VIDEO	MP4, YouTube, Vimeo
PDF	Documentos, manuales
AUDIO	Podcasts, narraciones
IMAGE	PNG, JPG, SVG
LINK	Enlaces externos
DOCUMENT	Word, Excel
INTERACTIVE	Actividades interactivas
Ejemplo en código:
json
{

🤝 CONTRIBUCIONES (CONTRIBUTION)
¿Qué es?
Permite a los usuarios colaborar en la mejora del contenido.

Tipos de contribuciones:
Tipo	Descripción
SUGGESTION	Sugerencia de mejora
CORRECTION	Corrección de errores
TRANSLATION	Traducción a otro idioma
RESOURCE	Aporte de recursos
FEEDBACK	Retroalimentación
Ejemplo en código:
json
{
  "id": "contrib-001",
  "subtopicId": "subtopic-001",
  "userId": "user-321",
  "type": "SUGGESTION",
  "content": "Sugiero agregar un video más detallado sobre los periféricos",
  "status": "PENDING"
}
Estados de contribución:
PENDING: Pendiente de revisión

REVIEWED: Revisado

ACCEPTED: Aceptado

REJECTED: Rechazado

text

text
